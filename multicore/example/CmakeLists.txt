cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2)
set(PICO_PLATFORM rp2350-arm-s)
include(pico_sdk_import.cmake)
project(multicore C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()

add_executable(multicore  multicore.c)

pico_enable_stdio_usb(multicore 1)
pico_enable_stdio_uart(multicore 0)   # keep Pico UART free for the GPS

target_link_libraries(multicore pico_stdlib pico_multicore hardware_uart hardware_spi)

pico_add_extra_outputs(multicore)
